2. Get the application URL:
{{- if .Values.ingress.enabled }}
  {{- range .Values.ingress.hosts }}
   http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}
  {{- end }}
{{- else }}
   export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} \
     -l "app.kubernetes.io/name={{ include "horilla-ph.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" \
     -o jsonpath="{.items[0].metadata.name}")
   kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8000:8000
   # Then open http://127.0.0.1:8000
{{- end }}
